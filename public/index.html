Drop-in SDK Integration

Wrap standard AI tool calls with GuardAgent gateway checks in 3 lines of code.

PYTHON (LANGCHAIN / CREWAI) TYPESCRIPT / NODE.JS RAW REST PROXY
sdk_version: 1.4.2 
from guardagent import GuardAgentGateway, PolicyViolationError

# Initialize authoritative gateway client
guard = GuardAgentGateway(api_key="ga_live_7a9f81...", endpoint="https://gateway.guardagent.internal:8443")

@tool
def execute_agent_transfer(account_id: str, amount_usd: float):
    """Enforces zero-trust financial boundaries before calling banking ledger."""
    decision = guard.evaluate(
        agent_id="agt-finance-core-01",
        tool_name="bank_wire_transfer",
        parameters={"account_id": account_id, "amount_usd": amount_usd}
    )
    
    if decision.is_blocked:
        raise PolicyViolationError(f"Action blocked by policy: {decision.rationale}")
        
    if decision.requires_approval:
        return f"Escrow transfer paused. Awaiting human approval token [ID: {decision.request_id}]"

    return banking_service.dispatch(account_id, amount_usd)
