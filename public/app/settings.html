Configure your autonomous agents to route tool invocations through the GuardAgent proxy: 

import requests

def execute_agent_tool(agent_id, tool_name, action_name, parameters):
    url = "http://localhost:3000/api/requests/evaluate"
    headers = { "Content-Type": "application/json" }
    payload = {
        "agent_id": agent_id,
        "tool_name": tool_name,
        "action_name": action_name,
        "parameters": parameters
    }
    response = requests.post(url, json=payload, headers=headers).json()
    
    if response["decision"] == "ALLOW":
        return response["execution"]["result"]
    elif response["decision"] == "REQUIRE_APPROVAL":
        print(f"Action escalated to Human-in-the-Loop review queue. Request ID: {response['request_id']}")
        return None
    else:
        raise PermissionError(f"Action blocked by GuardAgent: {response['rationale']}")
